2. Sampel CustomPainter Kolaj Flutter (`collage_painter.dart`):

class CollagePainter extends CustomPainter {
  final List<ui.Image> images;
  final CollageTemplate template;
  final double spacing;
  final double borderRadius;

  CollagePainter({required this.images, required this.template, required this.spacing, required this.borderRadius});

  @override
  void paint(Canvas canvas, Size size) {
    // Canvas Paint Logic for 20+ Layout Masks
    for (int i = 0; i < template.slots.length; i++) {
      if (i >= images.length) break;
      final slot = template.slots[i].getRect(size, spacing);
      final RRect rrect = RRect.fromRectAndRadius(slot, Radius.circular(borderRadius));
      
      canvas.save();
      canvas.clipRRect(rrect);
      // Auto-center based on face focal point
      paintImage(canvas: canvas, rect: slot, image: images[i], fit: BoxFit.cover);
      canvas.restore();
    }
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => true;
}
